cmake_minimum_required(VERSION 3.16)
project(vrvideo CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(PkgConfig REQUIRED)
pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv)
pkg_check_modules(OPENXR REQUIRED IMPORTED_TARGET openxr)
pkg_check_modules(GLFW REQUIRED IMPORTED_TARGET glfw3)
find_package(Vulkan REQUIRED)
find_package(OpenGL REQUIRED)

add_executable(vrvideo src/main.cpp src/gl.cpp)
target_link_libraries(vrvideo PRIVATE PkgConfig::MPV PkgConfig::OPENXR PkgConfig::GLFW Vulkan::Vulkan OpenGL::GL)
if(NOT MSVC)
  target_compile_options(vrvideo PRIVATE -Wall -Wno-missing-field-initializers)
endif()
install(TARGETS vrvideo RUNTIME DESTINATION bin)
